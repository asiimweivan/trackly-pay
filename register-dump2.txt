/**
 * authService.ts
 * Trackly - Auth API calls
 *
 * Fixed several request-shape mismatches against the real backend:
 * - register() never sent email at all, even though it's now mandatory
 * - sendOtp()/verifyOtp() sent {phone, countryCode} but backend expects {phone, purpose}
 * - resetPassword() sent {phone, countryCode, otp} but backend expects {email, otpCode}
 *   (password reset is now email-based, sent via Brevo, since no SMS/WhatsApp
 *   provider is configured yet)
 */
import { api, tokenStorage } from './api';

export interface User {
  id:         string;
  phone:      string;
  fullName:   string;
  email:      string | null;
  role:       string;
  appRole?:   string;
  businessId?: string;
  isVerified: boolean;
  pinEnabled: boolean;
}

export interface AuthResponse {
  accessToken:  string;
  refreshToken: string;
  user:         User;
}

export const authService = {
  async register(phone: string, fullName: string, email: string, password: string, role: string = 'business_owner'): Promise<AuthResponse> {
    const res = await api.post<AuthResponse>('/auth/register', { phone, fullName, email, password, role });
    await tokenStorage.setTokens(res.accessToken, res.refreshToken);
    await tokenStorage.setUser(res.user);
    return res;
  },

  async login(phone: string, password: string): Promise<AuthResponse> {
    const res = await api.post<AuthResponse>('/auth/login', { phone, password });
    await tokenStorage.setTokens(res.accessToken, res.refreshToken);
    await tokenStorage.setUser(res.user);
    return res;
  },

  async pinLogin(phone: string, pin: string): Promise<AuthResponse> {
    const res = await api.post<AuthResponse>('/auth/pin/login', { phone, pin });
    await tokenStorage.setTokens(res.accessToken, res.refreshToken);
    await tokenStorage.setUser(res.user);
    return res;
  },

  async setupPin(pin: string): Promise<void> {
    await api.post('/auth/pin/setup', { pin });
  },

  // For phone verification purposes (register/verify_phone) -- SMS-based,
  // still logs to console server-side until an SMS provider is configured.
  async sendOtp(phone: string, purpose: 'register' | 'verify_phone'): Promise<void> {
    await api.post('/auth/otp/send', { phone, purpose });
  },

  async verifyOtp(phone: string, purpose: 'register' | 'verify_phone', code: string): Promise<void> {
    await api.post('/auth/otp/verify', { phone, purpose, code });
  },

  // Forgot password is email-based -- code is emailed via Brevo.
  async forgotPassword(email: string): Promise<void> {
    await api.post('/auth/password/forgot', { email });
  },

  async resetPassword(email: string, otpCode: string, newPassword: string): Promise<void> {
    await api.post('/auth/password/reset', { email, otpCode, newPassword });
  },

  // Permanently deletes the account and all business data (password required).
  async deleteAccount(password: string): Promise<void> {
    await api.post('/auth/account/delete', { password });
    await tokenStorage.clear();
  },

  // A deleted account asks Trackly to recover it (phone + password prove ownership).
  async requestRecovery(phone: string, password: string): Promise<void> {
    await api.post('/auth/account/recovery-request', { phone, password });
  },

  async getMe(): Promise<User> {
    return api.get<User>('/auth/me');
  },

  async logout(): Promise<void> {
    await tokenStorage.clear();
  },
};

import { api } from './api';

export interface Business {
  id:             string;
  ownerId:        string;
  name:           string;
  category:       string;
  province:       string;
  district:       string;
  sector:         string | null;
  cell:           string | null;
  village:        string | null;
  phone:          string;
  email:          string | null;
  logoUrl:        string | null;
  currency:       string;
  timezone:       string;
  tinNumber:      string | null;
  isVatRegistered:boolean;
  annualTurnover: number;
  createdAt:      string | null;
}

export interface CreateBusinessRequest {
  name:            string;
  category:        string;
  province:        string;
  district:        string;
  sector?:         string;
  cell?:           string;
  village?:        string;
  phone:           string;
  email?:          string;
  tinNumber?:      string;
  referralCode?:   string;
  isVatRegistered: boolean;
  annualTurnover:  number;
}

export const businessService = {

  async create(req: CreateBusinessRequest): Promise<Business> {
    return api.post<Business>('/businesses', req);
  },

  async getAll(): Promise<Business[]> {
    return api.get<Business[]>('/businesses');
  },

  async getById(id: string): Promise<Business> {
    return api.get<Business>(`/businesses/${id}`);
  },

  async update(id: string, req: Partial<CreateBusinessRequest>): Promise<Business> {
    return api.patch<Business>(`/businesses/${id}`, req);
  },

  async delete(id: string): Promise<void> {
    return api.delete(`/businesses/${id}`);
  },
};
